void main() {
  var years = [1705, 1900, 1601, 2000]; // года
  
  for (var year in years){ // цикл на все переменные списка
    double res = year / 100; // делю чтобы получились целые числа года
    int intYear = res.round(); // округляю чтобы убрать сотые
    var wek = intYear; // век
    //print("$res; $intYear");
    
    if (res > intYear){ // если целое число больше чем округлённое, 
        wek +=1; // то прибавляется к веку
    }
    
    print('Год: $year ; Век = $wek'); // вывод
  }
}
